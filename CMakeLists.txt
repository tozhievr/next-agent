cmake_minimum_required(VERSION 4.5)
project(NextAgent VERSION 0.44.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 23.1.3)
  message(FATAL_ERROR "NextAgent requires Clang 23.1.3 or newer")
endif()

set(LX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/next-langserver/source")
set(LX_TESTS OFF CACHE BOOL "" FORCE)
include_directories("${LX_SOURCE}" "${LX_SOURCE}/include" "${LX_SOURCE}/engine/rope")
add_subdirectory("${LX_SOURCE}" langserver EXCLUDE_FROM_ALL)
foreach(target lx_common lx_atn lx_rope lx_uast lx_lex lx_parse lx_host lx_host_services)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
endforeach()

set(NEXTAGENT_SOURCES
  src/checks.cpp src/util.cpp src/json.cpp src/platform.cpp src/core.cpp src/store.cpp src/symbols.cpp
  src/lsp_proto.cpp src/compilation.cpp src/diagnostic_gate.cpp
  src/language_engine.cpp src/refactor.cpp src/index.cpp src/semantic.cpp
  src/hybrid.cpp src/embed.cpp src/merge.cpp src/rpc.cpp)
add_library(nextagent_core STATIC ${NEXTAGENT_SOURCES})
target_include_directories(nextagent_core PRIVATE src PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(nextagent_core PUBLIC _CRT_SECURE_NO_WARNINGS)
target_compile_options(nextagent_core PRIVATE -pedantic-errors)
target_link_libraries(nextagent_core PUBLIC lx_host)
if(WIN32)
  target_link_libraries(nextagent_core PUBLIC kernel32 ws2_32)
else()
  target_link_libraries(nextagent_core PUBLIC pthread dl)
endif()
add_executable(nextagent src/main.cpp)
target_link_libraries(nextagent PRIVATE nextagent_core)
target_compile_options(nextagent PRIVATE -pedantic-errors)
add_executable(nextagent_tests tests/tests.cpp)
target_link_libraries(nextagent_tests PRIVATE nextagent_core)
target_compile_options(nextagent_tests PRIVATE -pedantic-errors)
set_target_properties(nextagent_tests PROPERTIES OUTPUT_NAME tests)

if(WIN32)
  set(NEXTAGENT_PLATFORM windows)
  set(CPACK_GENERATOR ZIP)
else()
  set(NEXTAGENT_PLATFORM linux)
  set(CPACK_GENERATOR TGZ)
endif()
set(PLUGIN_BUNDLE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/next-langserver/${NEXTAGENT_PLATFORM}-x64/plugins")
if(NOT IS_DIRECTORY "${PLUGIN_BUNDLE}")
  message(FATAL_ERROR "Native plugins missing: run tools/build_plugins.py before configuring")
endif()
add_custom_target(nextagent_plugins ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${PLUGIN_BUNDLE}" "${CMAKE_CURRENT_BINARY_DIR}/plugins")
add_dependencies(nextagent nextagent_plugins)
add_dependencies(nextagent_tests nextagent_plugins)
include(CTest)
add_test(NAME unit COMMAND nextagent_tests)
install(TARGETS nextagent RUNTIME DESTINATION .)
install(DIRECTORY "${PLUGIN_BUNDLE}/" DESTINATION plugins)
install(FILES README.md LICENSE CHANGELOG.md docs/AGENT_GUIDE.md docs/HARNESS.md DESTINATION .)
set(CPACK_PACKAGE_NAME nextagent)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_PACKAGE_FILE_NAME "nextagent-${PROJECT_VERSION}-${NEXTAGENT_PLATFORM}-x64")
include(CPack)

add_executable(nextagent_core_regressions tests/core_regressions.cpp)
target_link_libraries(nextagent_core_regressions PRIVATE nextagent_core)
add_test(NAME core_regressions COMMAND nextagent_core_regressions)

add_library(nextagent_harness_core STATIC harness/http.cpp harness/runner.cpp)
target_compile_options(nextagent_harness_core PRIVATE -pedantic-errors)
target_include_directories(nextagent_harness_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(nextagent_harness_core PUBLIC nextagent_core)
if(WIN32)
  target_link_libraries(nextagent_harness_core PRIVATE winhttp crypt32)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(nextagent_harness_core PRIVATE CURL::libcurl)
endif()
add_executable(nextagent_harness_cli harness/cli.cpp)
set_target_properties(nextagent_harness_cli PROPERTIES OUTPUT_NAME nextagent-harness-cli)
target_link_libraries(nextagent_harness_cli PRIVATE nextagent_harness_core)
add_dependencies(nextagent_harness_cli nextagent_plugins)
add_executable(nextagent_harness_tests tests/harness_tests.cpp)
target_link_libraries(nextagent_harness_tests PRIVATE nextagent_harness_core)
add_dependencies(nextagent_harness_tests nextagent_plugins)
add_test(NAME harness COMMAND nextagent_harness_tests)
install(TARGETS nextagent_harness_cli RUNTIME DESTINATION .)

if(WIN32)
  set(AUI "${CMAKE_CURRENT_SOURCE_DIR}/vendor/arcade_ui_standalone/library")
  add_library(nextagent_arcade_ui STATIC
    ${AUI}/src/core/layout.cpp ${AUI}/src/core/host.cpp ${AUI}/src/core/text_edit.cpp
    ${AUI}/src/core/components.cpp ${AUI}/src/core/canvas.cpp
    ${AUI}/src/win32/text_engine.cpp ${AUI}/src/dx12/ui_renderer.cpp)
  target_include_directories(nextagent_arcade_ui PUBLIC ${AUI}/include)
  target_compile_definitions(nextagent_arcade_ui PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(nextagent_arcade_ui PUBLIC dwrite user32 imm32 d3d12 d3dcompiler d2d1 windowscodecs ole32)
  add_executable(nextagent_harness WIN32 harness/ui_win32.cpp)
  set_target_properties(nextagent_harness PROPERTIES OUTPUT_NAME nextagent-harness)
  target_link_options(nextagent_harness PRIVATE "LINKER:/ENTRY:mainCRTStartup")
  target_link_libraries(nextagent_harness PRIVATE nextagent_harness_core nextagent_arcade_ui dxgi shell32 comdlg32 psapi)
  add_dependencies(nextagent_harness nextagent_plugins)
  install(TARGETS nextagent_harness RUNTIME DESTINATION .)
endif()

find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME harness_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/harness_http.py $<TARGET_FILE:nextagent_harness_cli>)
set_tests_properties(harness_http PROPERTIES TIMEOUT 60)
if(WIN32)
 add_test(NAME harness_gui COMMAND nextagent_harness --smoke --warp --state-dir ${CMAKE_CURRENT_BINARY_DIR}/gui-state --screenshot ${CMAKE_CURRENT_BINARY_DIR}/harness-ui.bmp)
 set_tests_properties(harness_gui PROPERTIES TIMEOUT 60)
endif()

add_test(NAME native_languages COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_languages.py --executable $<TARGET_FILE:nextagent> --report build/${NEXTAGENT_PLATFORM}/native-language-test.json)
set_tests_properties(native_languages PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} TIMEOUT 120)
